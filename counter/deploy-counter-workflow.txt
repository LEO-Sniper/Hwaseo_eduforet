name: Deploy visitor counter
on:
  workflow_dispatch:
permissions:
  contents: read
concurrency:
  group: eduforet-counter-deploy
  cancel-in-progress: false
jobs:
  deploy:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v6
      - uses: actions/setup-node@v4
        with:
          node-version: '24'
      - name: Verify visitor counting
        run: node counter/test.mjs
      - name: Deploy shared visitor storage
        uses: cloudflare/wrangler-action@v4
        with:
          apiToken: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          accountId: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
          command: deploy --config counter/wrangler.jsonc
